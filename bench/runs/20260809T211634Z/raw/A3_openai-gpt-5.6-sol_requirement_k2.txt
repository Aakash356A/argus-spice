* LT3094 negative LDO: -7V input, -5V output
.include LT3094.sub

VINPUT IN 0 -7

CIN IN 0 10u
COUT OUT 0 22u

RSET SET 0 49.9k
CSET SET 0 10n

RENUV ENUV IN 1
RILIM ILIM 0 10k
RPG PG 0 10k

RLOAD OUT 0 10k

XU1 IN OUT SET 0 ENUV ILIM PG OUT LT3094

.tran 0 100m 0 10u startup
.meas tran vout_dc FIND V(OUT) AT=90m

.end
