* LTC6252 non-inverting amplifier, gain = 2 V/V
.include LTC6252.sub

VCC VPLUS 0 5
VEE VMINUS 0 -5
VIN IN 0 DC 0 AC 1 SINE(0 1 10Meg)

XU1 IN NFB VPLUS VMINUS OUT LTC6252
RG NFB 0 1k
RF OUT NFB 1k
RLOAD OUT 0 10k

.options plotwinsize=0
.tran 0 2u 1u 0.2n
.ac dec 100 1Meg 100Meg

.meas tran VIN_RMS RMS V(IN) FROM 1u TO 2u
.meas tran VOUT_RMS RMS V(OUT) FROM 1u TO 2u
.meas tran gain_db PARAM 20*log10(VOUT_RMS/VIN_RMS)
.meas ac gain_db_ac FIND db(V(OUT)/V(IN)) AT=10Meg

.end
