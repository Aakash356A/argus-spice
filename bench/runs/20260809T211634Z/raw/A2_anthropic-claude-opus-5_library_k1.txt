* ADP7118 Low-Dropout Regulator: 3.3 V IN -> 1.8 V OUT @ 50 mA
* -----------------------------------------------------------
* Vendor macro-model library (must be on the LTspice search path)
.lib ADP7102.lib

* ---------------- Source and input decoupling ----------------
VIN   IN  0  DC 3.3
CIN   IN  0  1u                  ; input decoupling cap (X5R/X7R)
RESR_IN CIN_X 0 1m               ; (unused helper, keeps ESR modelling simple)

* ---------------- LDO: IN OUT ADJ EN GND ---------------------
* EN is tied to IN so the regulator is permanently enabled.
XU1   IN  OUT  ADJ  IN  0  ADP7118

* ---------------- Output-voltage programming -----------------
* VOUT = VREF * (1 + R1/R2), VREF = 1.20 V
* 1.20 * (1 + 10k/20k) = 1.80 V
R1    OUT ADJ 10k
R2    ADJ 0   20k
CFF   OUT ADJ 10p               ; small feed-forward cap for stability

* ---------------- Output decoupling and load -----------------
COUT  OUT 0  1u
RESR  OUT 0  1G                 ; keeps OUT node referenced at t=0
RLOAD OUT 0  36                 ; 1.8 V / 36 ohm = 50 mA

* ---------------------- Analysis -----------------------------
.tran 0 5m 0 5u
.meas TRAN vout_dc AVG V(OUT) FROM=4m TO=5m
.meas TRAN iload   AVG I(RLOAD) FROM=4m TO=5m
.options plotwinsize=0 gmin=1e-12 reltol=1e-4

.end
