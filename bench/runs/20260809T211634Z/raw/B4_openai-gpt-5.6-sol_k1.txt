* AD8551 non-inverting amplifier, gain = 100 V/V
VCC VCC 0 5
VIN IN 0 SINE(25m 10m 1k) AC 10m

RFB OUT NFB 99k
RG  NFB 0 1k

XU1 IN NFB VCC 0 OUT AD8551

.ac dec 100 10 1Meg
.tran 0 5m 0 2u
.meas AC gain_db FIND (VDB(OUT)-VDB(IN)) AT=1k

.subckt AD8551 VP VN VDD VSS VOUT
GERR VSS NDOM VP VN 1m
RDOM NDOM VSS 100Meg
CDOM NDOM VSS 106.1p
BOUT VOUT VSS V=limit(V(NDOM)-V(VSS),5m,V(VDD)-V(VSS)-5m)
RLOAD VOUT VSS 100Meg
.ends AD8551

.end
